case 'utterance': {
  const msg: Message = {
    id: `${Date.now()}-${Math.random()}`,
    sender: event.sender, text: event.text,
    ts: Date.now(), media: event.media,
  }
  setNodes(nds => nds.map(n => {
    if (n.type === 'AgentNode' && asAgent(n.data).name === event.sender)
      return { ...n, data: { ...n.data, messages: [...n.data.messages, msg] } }
    if (n.type === 'HumanNode' && asHuman(n.data).humanName === event.sender)
      return { ...n, data: { ...n.data, messages: [...n.data.messages, msg] } }
    return n
  }))
  // Image utterance → ImageArtifactNode
  if (event.media?.type === 'image') {
    const artifactId = `artifact-img-${Date.now()}`
    setNodes(nds => [...nds, {
      id: artifactId, type: 'ImageArtifactNode',
      position: nextArtifactPosition(nds, `agent-${event.sender}`),
      data: { kind: 'image', agentName: event.sender,
              label: 'image', preview: event.media!.url, url: event.media!.url },
    }])
    setEdges(eds => [...eds, glowEdge(
      `e-agent-${event.sender}-${artifactId}`,
      `agent-${event.sender}`, artifactId,
      { color: '#8b5cf6', dashed: true },
    )])
  }
  break
}

case 'floor_grant': {
  setNodes(nds => nds.map(n => {
    if (n.type !== 'AgentNode') return n
    const nd = asAgent(n.data)
    const state: AgentFloorState =
      nd.name === event.to ? 'speaking' : 'waiting'

    ...e,

      ...asGlowEdge(e.data),
      active: e.source === FLOOR_ID && e.target === targetId,

floor_yield

OFP yieldFloor → teal flash → auto-reverts after 1.5s
{ "type": "floor_yield", "from": "Alice", "from_uri": "tag:ofp-playground.local,2025:llm-Alice" }
• Transient state (1.5 s): floorState = 'yielding'
• Edge pulses teal (solid flash fading out)
• After 1.5 s: reverts to 'waiting', edge pulse cleared
• Signals agent voluntarily handed back the floor
ANTHROPIC Alice ⏳ waiting
App.tsx handler
case 'floor_yield': {
  const targetId = `agent-${event.from}`
  setNodes(nds => nds.map(n => {
    if (n.type !== 'AgentNode') return n
    const nd = asAgent(n.data)
    if (nd.name !== event.from) return n
    return { ...n, data: { ...nd, floorState: 'yielding' } }
  }))
  setEdges(eds => eds.map(e =>
    e.source === FLOOR_ID && e.target === targetId
      ? { ...e, data: { ...asGlowEdge(e.data), active: false, pulse: 'teal' } }
      : e
  ))
  setTimeout(() => {
    // revert after 1500ms
    setNodes(/* floorState = 'waiting' */)
    setEdges(/* pulse = null */)
  }, 1500)
  break
}

App.tsx — artifact routing
function artifactNodeType(kind: string): string {
  if (kind === 'image') return 'ImageArtifactNode'
  if (kind === 'code' || kind === 'file') return 'CodeArtifactNode'
  if (kind === 'music') return 'MusicArtifactNode'
  return 'PhaseArtifactNode'
}

const ARTIFACT_COLOR: Record<string, string> = {
  image:    '#8b5cf6',
  code:     '#10a37f',
  file:     '#10a37f',
  phase:    '#58a6ff',
  markdown: '#58a6ff',
  music:    '#f97316',
}

case 'artifact_saved': {
  const agentId = `agent-${event.agent}`
  const artId = `artifact-${event.slug ?? Date.now()}`
  const color = ARTIFACT_COLOR[event.kind] ?? '#58a6ff'
  setNodes(nds => [...nds, {
    id: artId, type: artifactNodeType(event.kind),
    position: nextArtifactPosition(nds, agentId),
    data: { kind: event.kind, agentName: event.agent,
            slug: event.slug, label: event.slug ?? 'artifact',
            preview: event.preview },
  }])
  setEdges(eds => [...eds,
    glowEdge(`e-${agentId}-${artId}`, agentId, artId,
             { color, dashed: true })
  ])
  break
}

  agentName: string
  slug?: string
  label: string     // filename or slug
  preview: string   // first ~120 chars, or image URL
  url?: string      // direct URL for images / audio

// Event → node type routing (App.tsx)
function artifactNodeType(kind: string): string {
  if (kind === 'image')              return 'ImageArtifactNode'
  if (kind === 'code' || kind==='file') return 'CodeArtifactNode'
  if (kind === 'music')              return 'MusicArtifactNode'
  return 'PhaseArtifactNode'
}

// Position: right of producing agent, staggered vertically
function nextArtifactPosition(nodes, agentNodeId) {
  const agent = nodes.find(n => n.id === agentNodeId)
  const count = nodes.filter(n =>
    n.type?.includes('ArtifactNode') &&
    n.data.agentName === agent?.data?.name
  ).length
  return { x: (agent?.position.x ?? 400) + 260,
           y: (agent?.position.y ?? 100) + count * 140 }
}

GlowEdge States

canvas/frontend/src/edges/GlowEdge.tsx — custom ReactFlow edge with 5 visual states.

Inactive (default — Floor→Agent, no floor holder)
Active — floor granted to this agent (animated dash travel)
Dashed — artifact edge (Agent → ArtifactNode, never active)
Pulse: amber — floor_request (blinking dashed)
Pulse: teal — floor_yield (solid, fades out in 1.5s)
Pulse: purple — manifest_published (solid, fades out in 1.5s)
GlowEdge.tsx — full component
interface GlowEdgeData {
  active: boolean      // true while target agent holds floor
  color: string        // stroke color
  dashed: boolean      // true for artifact edges
  pulse?: 'amber' | 'teal' | 'purple' | null
}

export function GlowEdge({ id, sourceX,sourceY,targetX,targetY,
                            sourcePosition,targetPosition, data }) {
  const { color, active, dashed, pulse } = data as GlowEdgeData
  const [edgePath] = getBezierPath({...})

  return (
    <>
      {/* Base path */}
      <path d={edgePath} style={{
        stroke: color, strokeWidth: active ? 2.5 : 1.5,
        strokeDasharray: dashed ? '5 4' : undefined,
        fill: 'none',
      }} />

      {/* Active glow overlay */}
      {active && <path d={edgePath} style={{
        stroke: color, strokeWidth: 3,
        strokeDasharray: '10 8', fill: 'none',
        filter: `drop-shadow(0 0 5px ${color})`,
        animation: 'dashTravel 1.0s linear infinite',
      }} />}

      {/* Amber — requestFloor */}
      {pulse==='amber' && <path d={edgePath} style={{
        stroke:'#d4a843', strokeWidth:2,
        strokeDasharray:'6 5', fill:'none',
        filter:'drop-shadow(0 0 4px #d4a843)',
        animation:'edgeBlinkAmber 0.9s ease-in-out infinite',
      }} />}

      {/* Teal — yieldFloor */}
      {pulse==='teal' && <path d={edgePath} style={{
        stroke:'#00bcd4', strokeWidth:3, fill:'none',
        filter:'drop-shadow(0 0 6px #00bcd4)',
        animation:'ofp-edge-flash-teal 1.5s ease-out forwards',
      }} />}

      {/* Purple — publishManifest */}
      {pulse==='purple' && <path d={edgePath} style={{
        stroke:'#8b5cf6', strokeWidth:3, fill:'none',
        filter:'drop-shadow(0 0 6px #8b5cf6)',
        animation:'ofp-edge-flash-purple 1.5s ease-out forwards',
      }} />}
    </>
  )
}
